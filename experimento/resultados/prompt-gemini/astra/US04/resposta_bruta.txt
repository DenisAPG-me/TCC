## SUPOSIÇÕES

- Não houve acesso ao sistema ou à internet, nem execução dos testes. Os seletores e detalhes não especificados são premissas a validar.
- O formulário está na rota `/` da URL base `https://automationintesting.online/`, sem necessidade de autenticação.
- Os campos possuem rótulos acessíveis únicos `Name`, `Email`, `Phone`, `Subject` e `Message`, associados aos respectivos controles. O botão possui papel `button` e nome acessível exato `Submit`.
- As mensagens de obrigatoriedade são, por suposição: `Name may not be blank`, `Email may not be blank`, `Phone may not be blank`, `Subject may not be blank` e `Message may not be blank`. Esses textos deverão ser ajustados caso a aplicação use outras expressões.
- Os erros são apresentados como textos visíveis. O erro de e-mail pode incluir um prefixo; por isso, verifica-se a presença da expressão exigida pelo critério.
- A confirmação apresenta o texto exato `Thanks for getting in touch <nome>!` e o assunto em um elemento de texto próprio.
- Os intervalos mencionados nas mensagens de erro são inclusivos e definem também os limites máximos aceitos. Os casos acima dos máximos são derivados desses intervalos.
- Os controles permitem inserir valores acima dos limites e submeter e-mail malformado; validações nativas do navegador ou atributos `maxlength` não impedem observar as mensagens especificadas.
- Telefones compostos apenas por dígitos são válidos quando respeitam o comprimento. Não são presumidas regras adicionais de formato, DDD ou país.
- `Ana Silva`, `ana.silva@example.com`, telefone de 15 dígitos, assunto `Consulta sobre reserva` e a mensagem padrão do script são dados válidos. Os dados de borda usam caracteres ASCII para tornar inequívoca a contagem.
- Cada teste começa em um contexto isolado fornecido pelo plugin. O ambiente permite envios sucessivos com os mesmos dados.
- Não são adicionadas regras ausentes da User Story, como limites para nome, tratamento de espaços, formatos específicos de telefone ou persistência da mensagem.
- O requisito obrigatório autoriza `pytest-playwright`, além de `pytest` e `playwright`. Instalação: `python -m pip install pytest playwright pytest-playwright`; navegador: `python -m playwright install chromium`.
- Salve o arquivo como `test_contato.py` e execute `python -m pytest test_contato.py --browser chromium` ou `python test_contato.py`. Há exatamente uma função de teste por ID, sem parametrização.

```python
import sys

import pytest
from playwright.sync_api import Page, expect


BASE_URL = "https://automationintesting.online/"

PHONE_ERROR = "Phone must be between 11 and 21 characters."
SUBJECT_ERROR = "Subject must be between 5 and 100 characters."
MESSAGE_ERROR = "Message must be between 20 and 2000 characters."
EMAIL_ERROR = "must be a well-formed email address"

REQUIRED_ERRORS = (
    "Name may not be blank",
    "Email may not be blank",
    "Phone may not be blank",
    "Subject may not be blank",
    "Message may not be blank",
)

LABELS = {
    "name": "Name",
    "email": "Email",
    "phone": "Phone",
    "subject": "Subject",
    "message": "Message",
}

VALID_DATA = {
    "name": "Ana Silva",
    "email": "ana.silva@example.com",
    "phone": "119876543210123",
    "subject": "Consulta sobre reserva",
    "message": "Gostaria de saber se ha quartos disponiveis para reserva.",
}

expect.set_options(timeout=10_000)


def open_contact(page: Page) -> None:
    page.goto(BASE_URL, wait_until="domcontentloaded")
    expect(page.get_by_label("Name", exact=True)).to_be_visible()
    expect(
        page.get_by_role("button", name="Submit", exact=True)
    ).to_be_visible()


def submit_contact(page: Page, **overrides: str) -> dict:
    data = {**VALID_DATA, **overrides}
    open_contact(page)

    for field, label in LABELS.items():
        control = page.get_by_label(label, exact=True)
        control.fill(data[field])
        # Detecta truncamento dos dados, inclusive nos limites superiores.
        expect(control).to_have_value(data[field])

    page.get_by_role("button", name="Submit", exact=True).click()
    return data


def expect_error(page: Page, message: str, exact: bool = True) -> None:
    expect(page.get_by_text(message, exact=exact)).to_be_visible()


def expect_success(page: Page, data: dict) -> None:
    expect(
        page.get_by_text(
            f"Thanks for getting in touch {data['name']}!",
            exact=True,
        )
    ).to_be_visible()
    expect(
        page.get_by_text(data["subject"], exact=True)
    ).to_be_visible()


def test_CT01_formulario_vazio(page: Page) -> None:
    open_contact(page)
    for label in LABELS.values():
        expect(page.get_by_label(label, exact=True)).to_have_value("")

    page.get_by_role("button", name="Submit", exact=True).click()

    for message in REQUIRED_ERRORS:
        expect_error(page, message)


def test_CT02_telefone_10_caracteres(page: Page) -> None:
    submit_contact(page, phone="1" * 10)
    expect_error(page, PHONE_ERROR)


def test_CT03_assunto_e_mensagem_abaixo_dos_minimos(page: Page) -> None:
    submit_contact(page, subject="S" * 4, message="M" * 19)
    expect_error(page, SUBJECT_ERROR)
    expect_error(page, MESSAGE_ERROR)


def test_CT04_email_malformado(page: Page) -> None:
    submit_contact(page, email="bad")
    expect_error(page, EMAIL_ERROR, exact=False)


def test_CT05_dados_validos(page: Page) -> None:
    data = submit_contact(page)
    expect_success(page, data)


def test_CT06_telefone_11_caracteres(page: Page) -> None:
    data = submit_contact(page, phone="1" * 11)
    expect_success(page, data)


def test_CT07_telefone_21_caracteres(page: Page) -> None:
    data = submit_contact(page, phone="1" * 21)
    expect_success(page, data)


def test_CT08_telefone_22_caracteres(page: Page) -> None:
    submit_contact(page, phone="1" * 22)
    expect_error(page, PHONE_ERROR)


def test_CT09_assunto_4_caracteres(page: Page) -> None:
    submit_contact(page, subject="S" * 4)
    expect_error(page, SUBJECT_ERROR)


def test_CT10_assunto_5_caracteres(page: Page) -> None:
    data = submit_contact(page, subject="S" * 5)
    expect_success(page, data)


def test_CT11_assunto_100_caracteres(page: Page) -> None:
    data = submit_contact(page, subject="S" * 100)
    expect_success(page, data)


def test_CT12_assunto_101_caracteres(page: Page) -> None:
    submit_contact(page, subject="S" * 101)
    expect_error(page, SUBJECT_ERROR)


def test_CT13_mensagem_19_caracteres(page: Page) -> None:
    submit_contact(page, message="M" * 19)
    expect_error(page, MESSAGE_ERROR)


def test_CT14_mensagem_20_caracteres(page: Page) -> None:
    data = submit_contact(page, message="M" * 20)
    expect_success(page, data)


def test_CT15_mensagem_2000_caracteres(page: Page) -> None:
    data = submit_contact(page, message="M" * 2000)
    expect_success(page, data)


def test_CT16_mensagem_2001_caracteres(page: Page) -> None:
    submit_contact(page, message="M" * 2001)
    expect_error(page, MESSAGE_ERROR)


if __name__ == "__main__":
    raise SystemExit(pytest.main([__file__, *sys.argv[1:]]))
```
